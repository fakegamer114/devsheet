Python Cheatsheet

Variables

x = 5
y = "Hello"

Data Types

int, float, str, bool, list, dict

Lists

my_list = [1, 2, 3]
my_list.append(4)

Dictionaries

my_dict = {'key': 'value'}
my_dict['new_key'] = 'new_value'

If Statement

if x > 5:
  print("x is greater than 5")

For Loop

for i in range(5):
  print(i)

While Loop

while x < 10:
  x += 1

Function

def my_function():
  return "Hello"

Print Statement

print("Hello World")
print(f"x = {x}")

Input

name = input("Enter name: ")

Strings

my_string = "Hello"
print(len(my_string))
print(my_string.upper())

Tuples

my_tuple = (1, 2, 3)
print(my_tuple[0])

Sets

my_set = {1, 2, 3}
my_set.add(4)

List Comprehensions

squares = [x**2 for x in range(5)]
print(squares)

Dictionary Comprehensions

squares_dict = {x: x**2 for x in range(5)}
print(squares_dict)

Lambda Functions

add = lambda a, b: a + b
print(add(2, 3))

Classes

class MyClass:
  def __init__(self, value):
    self.value = value

Try-Except

try:
  x = 1 / 0
except ZeroDivisionError:
  print("Error")

File Handling

with open('file.txt', 'w') as f:
  f.write("Hello")

Modules

import math
print(math.sqrt(16))

List Slicing

my_list = [0, 1, 2, 3, 4]
print(my_list[1:3])
print(my_list[::2])

String Methods

text = "hello world"
print(text.upper())
print(text.split())

Enumerate

for i, item in enumerate(['a', 'b', 'c']):
  print(i, item)

Zip

names = ['Alice', 'Bob']
ages = [25, 30]
for name, age in zip(names, ages):
  print(name, age)

Map

numbers = [1, 2, 3]
squared = list(map(lambda x: x**2, numbers))

Filter

numbers = [1, 2, 3, 4, 5]
evens = list(filter(lambda x: x % 2 == 0, numbers))

*args

def func(*args):
  for arg in args:
    print(arg)
func(1, 2, 3)

**kwargs

def func(**kwargs):
  for key, value in kwargs.items():
    print(key, value)

Type Hints

def add(a: int, b: int) -> int:
  return a + b

Generator

def count_up(max):
  n = 1
  while n <= max:
    yield n
    n += 1

Decorators

def decorator(func):
  def wrapper():
    print("Before")
    func()
  return wrapper

Sorting

my_list = [3, 1, 2]
sorted_list = sorted(my_list)
my_list.sort()

List Methods

lst = [1, 2, 3]
lst.append(4)
lst.remove(2)
lst.pop()

Dictionary Methods

d = {'a': 1, 'b': 2}
print(d.keys())
print(d.values())
print(d.items())

Any & All

lst = [True, True, False]
print(any(lst))
print(all(lst))

Join & Split

lst = ['a', 'b', 'c']
text = '-'.join(lst)
words = text.split('-')

Context Managers

with open('file.txt', 'r') as f:
  content = f.read()

Inheritance

class Animal:
  def speak(self):
    pass
class Dog(Animal):
  def speak(self):
    return "Woof"

Magic Methods

class MyClass:
  def __init__(self, value):
    self.value = value
  def __str__(self):
    return str(self.value)

F-strings

name = "Alice"
age = 30
print(f"{name} is {age} years old")

Regular Expressions

import re
pattern = r'\d+'
text = "There are 123 apples"
matches = re.findall(pattern, text)

JSON Handling

import json
data = {'name': 'Alice', 'age': 30}
json_str = json.dumps(data)
parsed = json.loads(json_str)

Datetime

from datetime import datetime
now = datetime.now()
print(now.strftime("%Y-%m-%d"))

Random Module

import random
print(random.randint(1, 10))
print(random.choice(['a', 'b', 'c']))

OS Module

import os
print(os.getcwd())
os.mkdir('new_folder')

Recursion

def factorial(n):
  if n == 0:
    return 1
  return n * factorial(n-1)
