Python Cheatsheet

Variables

x = 5
y = "Hello"

Data Types

int, float, str, bool, list, dict

Lists

my_list = [1, 2, 3]
my_list.append(4)

Dictionaries

my_dict = {'key': 'value'}
my_dict['new_key'] = 'new_value'

If Statement

if x > 5:
  print("x is greater than 5")

For Loop

for i in range(5):
  print(i)

While Loop

while x < 10:
  x += 1

Function

def my_function():
  return "Hello"

Print Statement

print("Hello World")
print(f"x = {x}")

Input

name = input("Enter name: ")

Strings

my_string = "Hello"
print(len(my_string))
print(my_string.upper())

Tuples

my_tuple = (1, 2, 3)
print(my_tuple[0])

Sets

my_set = {1, 2, 3}
my_set.add(4)

List Comprehensions

squares = [x**2 for x in range(5)]
print(squares)

Dictionary Comprehensions

squares_dict = {x: x**2 for x in range(5)}
print(squares_dict)

Lambda Functions

add = lambda a, b: a + b
print(add(2, 3))

Classes

class MyClass:
  def __init__(self, value):
    self.value = value

Try-Except

try:
  x = 1 / 0
except ZeroDivisionError:
  print("Error")

File Handling

with open('file.txt', 'w') as f:
  f.write("Hello")

Modules

import math
print(math.sqrt(16))
